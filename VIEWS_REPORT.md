## 3. Text identity (acceptance bar item 3)

The script, run from the repository root as
`node text-identity.mjs ba16edd`:

```js
// text-identity.mjs: the .doc text of every page at the gate commit against the working tree.
// Usage: node text-identity.mjs <gate commit> ; run from the repository root.
import { execSync } from 'node:child_process';
import { readFileSync, existsSync } from 'node:fs';
const GATE = process.argv[2] || 'ba16edd';
const b = await import(process.cwd() + '/tools/build.mjs');
const text = (html) => { const d = b.docOf(html); return d ? b.textOf(d.html) : null; };
const files = execSync(`git ls-tree -r --name-only ${GATE} -- docs/`).toString().trim().split('\n').filter((f) => f.endsWith('.html') && !f.startsWith('docs/depmap/'));
let bad = 0;
for (const f of files) {
  const then = text(execSync(`git show ${GATE}:${f}`).toString());
  const nowPath = f === 'docs/index.html' ? 'docs/contents/index.html' : f;
  const now = existsSync(nowPath) ? text(readFileSync(nowPath, 'utf8')) : null;
  const ok = then === now;
  if (!ok) bad++;
  let detail = '';
  if (!ok && then && now && now.startsWith(then.slice(0, 200))) {
    let i = 0; while (i < then.length && then[i] === now[i]) i++;
    detail = ` first difference at char ${i}: now has "${now.slice(i, i + 160)}"`;
  }
  console.log(`${ok ? 'same' : 'DIFF'}  ${f}${nowPath !== f ? ' -> ' + nowPath : ''}${ok ? '' : ` (${then && then.length} vs ${now && now.length} chars)${detail}`}`);
}
console.log(bad ? `${bad} page(s) differ` : `every page identical (${files.length} pages)`);
```
